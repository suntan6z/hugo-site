{{- /* The tile beside a Now item. `logo` is one of: icon: a line icon from data/now_icons.json /img/… a file the site already serves from static/ a picture in the content/now/ bundle (SVG as is, others resized) and with none of those, the label's first letter. Takes (dict "logo" … "page" … "label" … "url" …). */ -}} {{- $page := .page -}} {{- $logo := .logo | default "" -}} {{- $class := "now-logo" -}} {{- $inner := "" -}} {{- if hasPrefix $logo "icon:" -}} {{- $class = "now-logo is-icon" -}} {{- $inner = partial "now-icon.html" (strings.TrimPrefix "icon:" $logo) -}} {{- else if hasPrefix $logo "/" -}} {{- $inner = printf `` $logo -}} {{- else if and $logo ($page.Resources.Get $logo) -}} {{- with $page.Resources.Get $logo -}} {{- if in (slice "jpeg" "png" "webp") .MediaType.SubType -}} {{- $img := .Fit "160x160 webp q85" -}} {{- $inner = printf `` $img.RelPermalink $img.Width $img.Height -}} {{- else -}} {{- $inner = printf `` .RelPermalink -}} {{- end -}} {{- end -}} {{- end -}} {{- if not $inner -}} {{- $class = "now-logo is-icon is-letter" -}} {{- $inner = substr (.label | default "•") 0 1 -}} {{- end -}} {{- if .url -}} {{ $inner | safeHTML }} {{- else -}} {{ $inner | safeHTML }} {{- end -}}
