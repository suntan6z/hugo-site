{{ $captionOverrides := dict }} {{ range hugo.Data.gallery_photos }} {{ $captionOverrides = merge $captionOverrides (dict .image_url (dict "caption" .caption "alt_text" .alt_text)) }} {{ end }} {{ $cityNames := dict }} {{ range hugo.Data.gallery_cities }} {{ $cityNames = merge $cityNames (dict .slug .name) }} {{ end }} {{ $photos := slice }} {{ $names := dict }} {{ range hugo.Data.gallery_cities }} {{ $city := .slug }} {{ with $.Site.GetPage (printf "/gallery/%s" $city) }} {{ range .Resources.ByType "image" }} {{ $filename := .Name }} {{ $imageURL := .RelPermalink }} {{ $digits := findRE "^[0-9]+" $filename }} {{ $num := 999999 }} {{ if gt (len $digits) 0 }}{{ $num = int (index $digits 0) }}{{ end }} {{ $sortkey := printf "%s_%06d_%s" $city $num $filename }} {{/* data/gallery_photos.json keys photos by their language-independent path (/gallery//), but .RelPermalink is language-prefixed (/en/gallery//) because defaultContentLanguageInSubdir is on. Looking up by RelPermalink alone silently matched nothing, so every caption and alt-text override in that file was dead. Try the canonical key first, then RelPermalink for any full-path entries. */}} {{ $overrideKey := printf "/gallery/%s/%s" $city $filename }} {{ $override := or (index $captionOverrides $overrideKey) (index $captionOverrides $imageURL) }} {{ $caption := "" }} {{ $altText := replace (i18n "photo_alt") "%city%" (or (index $cityNames $city) (title $city)) }} {{ if $override }} {{ with $override.caption }}{{ $caption = . }}{{ end }} {{ with $override.alt_text }}{{ $altText = . }}{{ end }} {{ end }} {{/* One readable name per photo, - (amsterdam-02), used for the three copies below and for the "Download original" file. */}} {{ $name := printf "%s-%s" $city (path.BaseName $filename) }} {{ if lt $num 999999 }}{{ $name = printf "%s-%02d" $city $num }}{{ end }} {{/* 2.jpg and 02.jpg would both be -02 and silently overwrite each other's copies: stop the build and say which files instead. */}} {{ with index $names $name }}{{ errorf "gallery: %s and %s in content/gallery/%s/ would both be published as %s; renumber one" . $filename $city $name }}{{ end }} {{ $names = merge $names (dict $name $filename) }} {{/* Three AVIF sizes, never the original (some are 8 MB): full_url — fits 1920px, what the lightbox settles on; preview_url — the same picture at 480px, shown the instant a photo is opened while full_url loads (same shape, so the swap doesn't jump); strip_url — a 2x crop of the strip's 17rem×22rem frame. The two small ones are made from the 1920px copy, not the original, so each multi-MB original is decoded once per build, not three times. Each is copied to /gallery//-.avif, so a visitor saving one gets amsterdam-02-1920.avif, not 2_hu_cf3e9b8e….webp. (Hugo rates AVIF q60 equal to WebP/JPEG q75; here it is ~20% smaller than the WebP q82 it replaced.) image_url stays the original's path: it is the key data/gallery_photos.json and the portal know the photo by. */}} {{ $dir := printf "gallery/%s" $city }} {{ $full := . }} {{ if or (gt .Width 1920) (gt .Height 1920) }}{{ $full = .Fit "1920x1920 avif q60" }}{{ else }}{{ $full = .Process "avif q60" }}{{ end }} {{ $preview := $full.Fit "480x480 avif q50" | resources.Copy (printf "%s/%s-480.avif" $dir $name) }} {{ $strip := $full.Fill "544x704 avif q55 smart" | resources.Copy (printf "%s/%s-strip.avif" $dir $name) }} {{ $full = $full | resources.Copy (printf "%s/%s-1920.avif" $dir $name) }} {{ $photos = $photos | append (dict "city" $city "name" $name "image_url" $imageURL "strip_url" $strip.RelPermalink "full_url" $full.RelPermalink "preview_url" $preview.RelPermalink "w" $full.Width "h" $full.Height "ow" .Width "oh" .Height "caption" $caption "alt_text" $altText "sortkey" $sortkey) }} {{ end }} {{ end }} {{ end }} {{ $photos = sort $photos "sortkey" }} {{ end }}
