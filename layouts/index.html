Looking up by RelPermalink alone silently matched nothing, so every caption and alt-text override in that file was dead. Try the canonical key first, then RelPermalink for any full-path entries. */}} {{ $overrideKey := printf "/gallery/%s/%s" $city $filename }} {{ $override := or (index $captionOverrides $overrideKey) (index $captionOverrides $imageURL) }} {{ $caption := "" }} {{ $altText := replace (i18n "photo_alt") "%city%" (or (index $cityNames $city) (title $city)) }} {{ if $override }} {{ with $override.caption }}{{ $caption = . }}{{ end }} {{ with $override.alt_text }}{{ $altText = . }}{{ end }} {{ end }} {{/* Three sizes, never the original (some are 8 MB): full_url — fits 1920px, what the lightbox settles on; preview_url — the same picture at 480px, shown the instant a photo is opened while full_url loads (same shape, so the swap doesn't jump); strip_url — a 2x crop of the strip's 17rem×22rem frame. The two small ones are made from the 1920px copy, not the original, so each multi-MB original is decoded once per build, not three times. image_url stays the original's path: it is the key data/gallery_photos.json and the portal know the photo by. */}} {{ $full := . }} {{ if or (gt .Width 1920) (gt .Height 1920) }}{{ $full = .Fit "1920x1920 webp q82" }}{{ else }}{{ $full = .Process "webp q82" }}{{ end }} {{ $preview := $full.Fit "480x480 webp q60" }} {{ $strip := $full.Fill "544x704 webp q78 smart" }} {{ $photos = $photos | append (dict "city" $city "image_url" $imageURL "strip_url" $strip.RelPermalink "full_url" $full.RelPermalink "preview_url" $preview.RelPermalink "w" $full.Width "h" $full.Height "ow" .Width "oh" .Height "caption" $caption "alt_text" $altText "sortkey" $sortkey) }} {{ end }} {{ end }} {{ end }} {{ $photos = sort $photos "sortkey" }} {{ end }}
